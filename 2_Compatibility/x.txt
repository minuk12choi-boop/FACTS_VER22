DELETE f
FROM facts_wip_source f
JOIN (
    SELECT
        lineid,
        processid,
        snap_date,
        MAX(loaded_at) AS max_loaded_at
    FROM facts_wip_source
    GROUP BY lineid, processid, snap_date
) latest
  ON f.lineid = latest.lineid
 AND f.processid = latest.processid
 AND f.snap_date = latest.snap_date
WHERE f.loaded_at < latest.max_loaded_at;
